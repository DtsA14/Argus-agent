"""Rational() must parse string arguments before combining them.

``Rational('0.5', '100')`` returned ``1/100100`` instead of ``1/200`` because
the string denominator was scaled with ``q *= p.q`` as a Python string
(``'100' * 2 == '100100'``) instead of as a number.
"""

from sympy import Rational


def _rational(*args, **kwargs):
    # Call Rational and report any exception as a failure of this test, so the
    # traceback shows which call raised it.
    try:
        return Rational(*args, **kwargs)
    except Exception as exc:
        raise AssertionError(
            "Rational(%s) raised %s: %s"
            % (", ".join(repr(a) for a in args), type(exc).__name__, exc)
        ) from exc


def test_rational_str_over_str():
    result = _rational('0.5', '100')
    assert result == Rational(1, 200), result
    assert result == Rational(0.5, 100), result


def test_rational_decimal_str_over_str():
    result = _rational('1.5', '2')
    assert result == Rational(3, 4), result


def test_rational_str_over_decimal_str():
    result = _rational('0.5', '0.25')
    assert result == Rational(2), result


def test_rational_str_over_negative_str():
    result = _rational('0.5', '-100')
    assert result == Rational(-1, 200), result
