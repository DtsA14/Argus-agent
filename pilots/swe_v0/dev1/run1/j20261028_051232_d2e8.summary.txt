==== sympy/core/tests/test_rational_two_string_args.py
"""Tests for Rational() built from two string arguments.

Rational('0.5', '100') must mean (1/2) / 100 == 1/200, exactly like
Rational(0.5, 100) does.  It used to give 1/100100 because the string
denominator was repeated by p.q ('100' * 2 -> '100100') instead of being
multiplied as a number.
"""

from sympy import Rational, S


def test_rational_two_string_arguments():
    assert Rational('0.5', '100') == Rational(1, 200)


def test_rational_two_string_arguments_agrees_with_float_argument():
    assert Rational('0.5', '100') == Rational(0.5, 100)


def test_rational_two_string_arguments_unreduced():
    assert Rational('1.5', '3') == Rational(1, 2)


def test_rational_numerator_rational_denominator_string():
    assert Rational(S.Half, '100') == Rational(1, 200)


def test_rational_two_fraction_strings():
    try:
        value = Rational('3/4', '1/2')
    except Exception as exc:
        raise AssertionError(
            "Rational('3/4', '1/2') raised %s: %s"
            % (type(exc).__name__, exc)) from exc
    assert value == Rational(3, 2)


def test_rational_integer_strings_unchanged():
    assert Rational('1', '100') == Rational(1, 100)
    assert Rational(1, '100') == Rational(1, 100)
    assert Rational('2', '4') == Rational(1, 2)


def test_rational_single_string_argument_unchanged():
    assert Rational('0.5') == S.Half
    assert Rational('1e-2/3.2') == Rational(1, 320)


def test_rational_invalid_string_denominator_still_raises():
    try:
        Rational('0.5', 'x')
    except (TypeError, ValueError):
        pass
    else:
        raise AssertionError("Rational('0.5', 'x') should raise")
