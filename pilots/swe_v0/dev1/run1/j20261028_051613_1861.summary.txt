from sympy import Piecewise, exp, sinh, symbols
from sympy.core.cache import clear_cache
from sympy.core.expr import Expr


def test_subs_int_to_float_piecewise_complex_symbols():
    # control: with plain (complex) symbols the substitution works today
    clear_cache()
    x, y, z = symbols('x y z')
    expr = exp(sinh(Piecewise((x, y > x), (y, True)) / z))
    res = expr.subs({1: 1.0})
    assert isinstance(res, Expr)
    assert res.has(Piecewise)


def test_subs_int_to_float_with_piecewise_hyperbolic():
    # substituting 1 -> 1.0 must not raise PolynomialError
    # ("Piecewise generators do not make sense") when the symbols are real
    clear_cache()
    x, y, z = symbols('x y z', real=True)
    expr = exp(sinh(Piecewise((x, y > x), (y, True)) / z))
    try:
        res = expr.subs({1: 1.0})
    except Exception as exc:
        raise AssertionError(
            'subs({1: 1.0}) raised %s: %s' % (type(exc).__name__, exc)) from exc
    assert isinstance(res, Expr)
    assert res.has(Piecewise)
    assert res.free_symbols == {x, y, z}
