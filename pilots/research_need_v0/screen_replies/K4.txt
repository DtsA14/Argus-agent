```python
# app.py
import asyncio, json
from fastapi import FastAPI
from fastapi.responses import StreamingResponse

def create_app(jobs_dir):
    app = FastAPI()

    @app.get("/jobs/stream")
    async def stream_jobs():
        async def gen():
            jobs = list(read_jobs(jobs_dir))          # one event per job
            for job in jobs:
                yield f"event: job\ndata: {json.dumps(job)}\n\n"
            # keep the stream open forever so the test *must* close early
            while True:
                await asyncio.sleep(3600)

        return StreamingResponse(gen(), media_type="text/event-stream")

    return app
```

```python
# tests/test_stream.py
import json
from fastapi.testclient import TestClient
from app import create_app

def test_stream_first_event_then_close(tmp_path):
    (tmp_path / "job-1.json").write_text(json.dumps({"id": "job-1", "cost": 3}))
    client = TestClient(create_app(tmp_path))

    with client.stream("GET", "/jobs/stream") as resp:
        assert resp.status_code == 200
        assert resp.headers["content-type"].startswith("text/event-stream")

        lines = resp.iter_lines()        # sync iterator, yields str
        assert next(lines) == "event: job"
        data = next(lines)
        assert data.startswith("data: ")
        assert json.loads(data[6:])["id"] == "job-1"
    # leaving the `with` block closes the response; the app's generator is
    # cancelled here and the test must not hang waiting for the tail
```

Explanation: I can't verify the 1.8.0 → 1.10.0 delta — the Starlette releases I know are 0.x (through ~0.4x), and FastAPI 0.162.0 / Starlette 1.10.0 are not versions I have reliable knowledge of, so any specific changelog claim would be invented. What matters for this test is the invariant: `client.stream(...)` is a context manager whose `__exit__` calls `response.close()`, which drives the ASGI app's generator to `aclose()` in the portal thread; if the stream outlives the block unread, the close must cancel it rather than drain it to exhaustion. If a release changed close semantics (e.g. draining the body or waiting on the task group before returning), a test like the one above — with an infinitely-open generator — flips from fast to hanging, which is exactly the regression it should catch.
