## The issue

TypeError: type 'peekable' is not subscriptable

It would be great if we could hint the type e.g. like:

```py
def f(it: peekable[str]):
    pass
```

Currently this works as workaround:
```py
def f(it: 'peekable[str]'):
    pass
```

```python
class PeekableSubscriptTests(TestCase):
    """Tests for subscripting ``peekable``, so it can be used in type hints
    written out at runtime, e.g. ``peekable[str]``.

    """

    def test_peekable_is_subscriptable_at_runtime(self):
        """Subscripting ``peekable`` must not raise ``TypeError``."""
        try:
            hint = mi.peekable[int]
        except TypeError as error:
            raise AssertionError(
                'peekable is not subscriptable: {!r}'.format(error)
            ) from error
        self.assertIsNotNone(hint)

    def test_subscripted_peekable_origin(self):
        """A subscripted ``peekable`` still refers to the same class."""
        try:
            hint = mi.peekable[str]
        except TypeError as error:
            raise AssertionError(
                'peekable is not subscriptable: {!r}'.format(error)
            ) from error
        self.assertIs(getattr(hint, '__origin__', hint), mi.peekable)

    def test_subscripted_peekable_still_works(self):
        """Objects made from a subscripted ``peekable`` behave as usual."""
        try:
            cls = mi.peekable[str]
        except TypeError as error:
            raise AssertionError(
                'peekable is not subscriptable: {!r}'.format(error)
            ) from error

        p = cls(['a', 'b'])
        self.assertIsInstance(p, mi.peekable)
        self.assertEqual(p.peek(), 'a')
        self.assertEqual(list(p), ['a', 'b'])
        self.assertEqual(list(mi.peekable[int](range(3))), [0, 1, 2])

```
