## The issue

Minor bug in numeric_range()

The code for `__reversed__` needs a special case for empty ranges.

An empty regular range, `list(reversed(range(0)))`, returns an empty list.  In contrast, an empty numeric range, `list(reversed(numeric_range(0)))` raises an `IndexError`.

## Test block X (to be added to tests/test_more.py)

```python
class NumericRangeEmptyReversedTests(TestCase):
    """Tests for ``reversed()`` on an empty ``numeric_range``"""

    def _check_reversed_is_empty(self, args, sl=None):
        r = mi.numeric_range(*args)
        if sl is not None:
            r = r[sl]
        self.assertEqual(0, len(r))
        self.assertFalse(bool(r))
        try:
            actual = list(reversed(r))
        except Exception as exc:
            raise AssertionError(
                f'list(reversed({r!r})) raised {exc!r} instead of '
                'returning an empty list'
            ) from exc
        self.assertEqual([], actual)

    def test_reversed_empty_range(self):
        for args in [
            (0,),
            (0.0,),
            (0, 0),
            (0.0, 5.0, -1.0),
            (Decimal('1.0'), Decimal('1.0')),
            (Fraction(1, 2), Fraction(1, 2), Fraction(1, 2)),
            (
                datetime(2019, 3, 29),
                datetime(2019, 3, 29),
                timedelta(hours=1),
            ),
        ]:
            self._check_reversed_is_empty(args)

    def test_reversed_empty_slice(self):
        for args, sl in [
            ((1.0, 9.0, 1.5), slice(2, 2)),
            ((1.0, 9.0, 1.5), slice(0, 0)),
            ((9.0, 1.0, -1.5), slice(-1, -1)),
        ]:
            self._check_reversed_is_empty(args, sl)
```
