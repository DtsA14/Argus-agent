## The issue

Decorator version of serialize()

**Proposal and rationale:**

The `serialize` tool wraps a single iterator instance and is used inline with instance creation: `serialize(itertools.count(100, 5))`.

If every instance needs to be serialized, it is more convenient to wrap the function that creates the iterator: `atomic_counter = synchonized(itertools.count)`.  Later, serialized instances can be created directly: `atomic_counter(100, 5)`.

The wrapper can also be used as a decorator around a generator definition.

**Draft implmentation:**

```python
def synchronized(func):
    """Wrap an function that produces iterators to serialized all instances.

    Existing itertools and more-itertools can be wrapped so that their
    instances are serialized.

    For example, ``itertools.count`` does not make thread-safe instances
    but that is easily fixed with::

        atomic_counter = synchronized(itertools.count)

    Can also be used as a decorator for generator functions definitions
    so that the generator instances are serialized::

        @synchronized
        def enumerate_and_timestamp(iterable):
            for count, value in enumerate(iterable):
                yield count, time_ns(), value                

    """
    @functools.wraps(func)
    def inner(*args, **kwargs):
        iterator = func(*args, **kwargs)
        return serialize(iterator)
    return inner
```

**Notes:**

The name "synchronized" came from Java.

```python
class TestSynchronized(TestCase):
    def test_concurrent_calls(self):
        unique = 10  # Number of distinct counters
        repetitions = 5  # Number of times each counter is used
        limit = 100  # Calls per counter per repetition

        @mi.synchronized
        def atomic_counter():
            # This is a generator so that non-concurrent calls are detectable.
            # To make calls while running more likely, this code uses random
            # time delays.
            i = 0
            while True:
                yield i
                next_i = i + 1
                sleep(random() / 1000)
                i = next_i

        def consumer(counter):
            for i in range(limit):
                next(counter)

        unique_counters = [atomic_counter() for _ in range(unique)]
        counters = unique_counters * repetitions
        workers = [
            Thread(target=consumer, args=[counter]) for counter in counters
        ]
        for worker in workers:
            worker.start()
        for worker in workers:
            worker.join()
        self.assertEqual(
            {next(counter) for counter in unique_counters},
            {limit * repetitions},
        )
```

```python
def _get_synchronized():
    """Return ``more_itertools.synchronized``, failing by assertion if absent."""
    try:
        return mi.synchronized
    except AttributeError as exc:
        raise AssertionError(
            'more_itertools.synchronized() is not defined'
        ) from exc


class TestSynchronized(TestCase):
    """Tests for ``synchronized()``"""

    def test_wrapped_itertool(self):
        """Instances made by the wrapper yield the itertool's values."""
        atomic_counter = _get_synchronized()(count)
        counter = atomic_counter(100, 5)
        self.assertEqual(
            [next(counter) for _ in range(4)], [100, 105, 110, 115]
        )

    def test_instances_are_independent(self):
        """Every call creates a fresh iterator."""
        atomic_counter = _get_synchronized()(count)
        first = atomic_counter()
        second = atomic_counter()
        self.assertEqual(next(first), 0)
        self.assertEqual(next(first), 1)
        self.assertEqual(next(second), 0)

    def test_decorated_generator_function(self):
        """The wrapper works as a decorator on a generator definition."""
        synchronized = _get_synchronized()

        @synchronized
        def enumerate_squares(iterable):
            'Yield the index and the square of each value.'
            for index, value in enumerate(iterable):
                yield index, value * value

        instance = enumerate_squares([2, 3])
        self.assertEqual(next(instance), (0, 4))
        self.assertEqual(list(instance), [(1, 9)])

    def test_decorator_preserves_metadata(self):
        """The wrapped function keeps its name and docstring."""
        synchronized = _get_synchronized()

        def producer(limit):
            'Docstring of the producer.'
            return iter(range(limit))

        wrapped = synchronized(producer)
        self.assertEqual(wrapped.__name__, 'producer')
        self.assertEqual(wrapped.__doc__, 'Docstring of the producer.')

    def test_arguments_are_forwarded(self):
        """Positional and keyword arguments reach the wrapped function."""
        synchronized = _get_synchronized()

        def repeat_value(value, times=2):
            for _ in range(times):
                yield value

        wrapped = synchronized(repeat_value)
        self.assertEqual(list(wrapped('a')), ['a', 'a'])
        self.assertEqual(list(wrapped(value='b', times=3)), ['b', 'b', 'b'])

    def test_concurrent_consumers(self):
        """Serialized instances are safe to share among threads."""
        result = 0
        result_lock = Lock()

        def producer(limit):
            'Non-concurrent producer. A generator version of range(limit).'
            for x in range(limit):
                yield x

        def consumer(counter):
            'Concurrent data consumer'
            nonlocal result
            total = 0
            for x in counter:
                total += x
            with result_lock:
                result += total

        limit = 10**5
        synchronized_producer = _get_synchronized()(producer)
        counter = synchronized_producer(limit)
        workers = [Thread(target=consumer, args=[counter]) for _ in range(10)]
        for worker in workers:
            worker.start()
        for worker in workers:
            worker.join()

        self.assertEqual(result, limit * (limit - 1) // 2)

```
