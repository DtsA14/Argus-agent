## Issue

Decorator version of serialize()

**Proposal and rationale:**

The `serialize` tool wraps a single iterator instance and is used inline with instance creation: `serialize(itertools.count(100, 5))`.

If every instance needs to be serialized, it is more convenient to wrap the function that creates the iterator: `atomic_counter = synchonized(itertools.count)`.  Later, serialized instances can be created directly: `atomic_counter(100, 5)`.

The wrapper can also be used as a decorator around a generator definition.

**Draft implmentation:**

```python
def synchronized(func):
    """Wrap an function that produces iterators to serialized all instances.

    Existing itertools and more-itertools can be wrapped so that their
    instances are serialized.

    For example, ``itertools.count`` does not make thread-safe instances
    but that is easily fixed with::

        atomic_counter = synchronized(itertools.count)

    Can also be used as a decorator for generator functions definitions
    so that the generator instances are serialized::

        @synchronized
        def enumerate_and_timestamp(iterable):
            for count, value in enumerate(iterable):
                yield count, time_ns(), value                

    """
    @functools.wraps(func)
    def inner(*args, **kwargs):
        iterator = func(*args, **kwargs)
        return serialize(iterator)
    return inner
```

**Notes:**

The name "synchronized" came from Java.

+from __future__ import annotations
+
+from itertools import count, islice
+from threading import Lock, Thread
+from unittest import TestCase
+
+import more_itertools as mi
+
+
+def _synchronized():
+    'Return ``more_itertools.synchronized``, failing an assert when absent.'
+    func = getattr(mi, 'synchronized', None)
+    assert callable(func), (
+        'more_itertools should provide a `synchronized` decorator that turns '
+        'a function producing iterators into a function producing '
+        '`serialize` instances'
+    )
+    return func
+
+
+class SynchronizedTests(TestCase):
+    def test_wraps_iterator_producer(self):
+        atomic_counter = _synchronized()(count)
+        iterator = atomic_counter(100, 5)
+        self.assertIsInstance(iterator, mi.serialize)
+        self.assertEqual(list(islice(iterator, 4)), [100, 105, 110, 115])
+        self.assertEqual(next(iterator), 120)
+
+    def test_each_call_creates_independent_iterator(self):
+        atomic_counter = _synchronized()(count)
+        first = atomic_counter(0, 1)
+        second = atomic_counter(0, 10)
+        self.assertIsNot(first, second)
+        self.assertEqual(next(first), 0)
+        self.assertEqual(next(first), 1)
+        self.assertEqual(next(second), 0)
+        self.assertEqual(next(second), 10)
+        self.assertEqual(next(first), 2)
+
+    def test_wraps_generator_function_as_decorator(self):
+        synchronized = _synchronized()
+
+        @synchronized
+        def enumerate_values(iterable):
+            for index, value in enumerate(iterable):
+                yield index, value
+
+        iterator = enumerate_values('abc')
+        self.assertIsInstance(iterator, mi.serialize)
+        self.assertEqual(list(iterator), [(0, 'a'), (1, 'b'), (2, 'c')])
+        self.assertEqual(
+            list(enumerate_values('xy')), [(0, 'x'), (1, 'y')]
+        )
+
+    def test_preserves_wrapped_function_metadata(self):
+        synchronized = _synchronized()
+
+        def counter(limit):
+            'Produce the numbers below limit.'
+            return iter(range(limit))
+
+        wrapper = synchronized(counter)
+        self.assertEqual(wrapper.__name__, 'counter')
+        self.assertEqual(wrapper.__doc__, 'Produce the numbers below limit.')
+        self.assertEqual(list(wrapper(3)), [0, 1, 2])
+
+    def test_concurrent_consumers_share_single_instance(self):
+        synchronized = _synchronized()
+        result = 0
+        result_lock = Lock()
+
+        @synchronized
+        def producer(limit):
+            'Non-concurrent producer. A generator version of range(limit).'
+            for x in range(limit):
+                yield x
+
+        def consumer(iterator):
+            'Concurrent data consumer'
+            nonlocal result
+            total = sum(iterator)
+            with result_lock:
+                result += total
+
+        limit = 10 ** 5
+        iterator = producer(limit)
+        workers = [Thread(target=consumer, args=[iterator]) for _ in range(10)]
+        for worker in workers:
+            worker.start()
+        for worker in workers:
+            worker.join()
+
+        self.assertEqual(result, limit * (limit - 1) // 2)
