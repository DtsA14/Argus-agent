## Issue

Minor bug in numeric_range()

The code for `__reversed__` needs a special case for empty ranges.

An empty regular range, `list(reversed(range(0)))`, returns an empty list.  In contrast, an empty numeric range, `list(reversed(numeric_range(0)))` raises an `IndexError`.

+"""Reversing an empty ``numeric_range`` must yield nothing.
+
+``list(reversed(range(0)))`` is ``[]``, and ``numeric_range`` is meant to
+behave like the built-in ``range``, so reversing an empty numeric range must
+return an empty iterator instead of raising ``IndexError``.
+"""
+
+from datetime import datetime
+from decimal import Decimal
+from fractions import Fraction
+
+import more_itertools as mi
+
+
+# Argument tuples that all describe an empty numeric range.
+EMPTY_ARGS = [
+    (0,),
+    (0.0,),
+    (Decimal('0.0'),),
+    (Fraction(0, 1),),
+    (datetime(2019, 3, 29), datetime(2019, 3, 29)),
+    (1.0, 1.0),
+    (1.0, 1.0, 1.5),
+    (0.0, 5.0, -1.0),
+    (5.0, 0.0, 1.0),
+    (Decimal('0.0'), Decimal('5.0'), Decimal('-1.5')),
+    (Fraction(5, 1), Fraction(0, 1), Fraction(1, 2)),
+]
+
+
+def test_reversed_of_empty_numeric_range_is_empty():
+    for args in EMPTY_ARGS:
+        description = 'numeric_range({})'.format(
+            ', '.join(repr(arg) for arg in args)
+        )
+        try:
+            actual = list(reversed(mi.numeric_range(*args)))
+        except Exception as exc:
+            raise AssertionError(
+                'reversed({}) raised {}: {} instead of yielding nothing'.format(
+                    description, type(exc).__name__, exc
+                )
+            ) from exc
+        assert actual == [], 'reversed({}) yielded {}'.format(
+            description, actual
+        )
+
+
+def test_reversed_of_numeric_range_sliced_to_nothing_is_empty():
+    empty = mi.numeric_range(0, 10, 2)[3:3]
+    try:
+        actual = list(reversed(empty))
+    except Exception as exc:
+        raise AssertionError(
+            'reversed({!r}) raised {}: {} instead of yielding nothing'.format(
+                empty, type(exc).__name__, exc
+            )
+        ) from exc
+    assert actual == [], 'reversed({!r}) yielded {}'.format(empty, actual)
