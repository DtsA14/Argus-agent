## The issue

TypeError: type 'peekable' is not subscriptable

It would be great if we could hint the type e.g. like:

```py
def f(it: peekable[str]):
    pass
```

Currently this works as workaround:
```py
def f(it: 'peekable[str]'):
    pass
```

    def test_class_getitem(self):
        """``peekable`` should be subscriptable at runtime"""
        try:
            subscripted = mi.peekable[int]
        except TypeError as e:
            raise AssertionError('peekable is not subscriptable') from e

        # The subscripted class should still build working peekables
        p = subscripted([1, 2, 3])
        self.assertEqual(p.peek(), 1)
        self.assertEqual(next(p), 1)
        self.assertEqual(list(p), [2, 3])

    def test_class_getitem_in_annotation(self):
        """A subscripted ``peekable`` should work as a type annotation"""

        class Annotated:
            it: mi.peekable[str]

        annotation = Annotated.__annotations__['it']
        # Annotations are strings here (``from __future__ import
        # annotations``), so evaluate this one the way Python would.
        try:
            annotation = eval(annotation)
        except TypeError as e:
            raise AssertionError('peekable[str] is not a usable annotation') from e

        self.assertIsNotNone(annotation)
