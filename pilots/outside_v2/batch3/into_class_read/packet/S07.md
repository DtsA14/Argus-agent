## The issue

split_before/split_after/split_when disagree with themselves on an empty iterable when maxsplit=0

`split_before`, `split_after` and `split_when` return a different result for an empty iterable when `maxsplit=0` than they do for any other `maxsplit`, even though no split can occur either way.

```python
>>> import more_itertools as mi
>>> pred = lambda x: x == 0
>>> {ms: list(mi.split_before([], pred, maxsplit=ms)) for ms in (-1, 0, 1, 2)}
{-1: [], 0: [[]], 1: [], 2: []}
```

`split_after` and `split_when` behave the same way. `split_at` is consistent (`[[]]` at every `maxsplit`, matching `''.split(',', 0) == ['']`).

The `[]` answer is the one the library already pins: `tests/test_more.py:1548` asserts `list(mi.split_before([], ...)) == []` and `tests/test_more.py:1689` asserts the same for `split_when`. Neither test passes `maxsplit`, so neither reaches the fast path that produces `[[]]`.

The fast path is:

```python
if maxsplit == 0:
    yield list(iterable)
    return
```

added to all four functions in f77823f (2020) as a speed shortcut, when `split_before` still ended with a bare `yield buf` and `[[]]` was its correct empty-input answer. 2e81a56 (2021) later changed the terminal yield to `if buf: yield buf` and added `test_empty_collection`, but did not reach behind the fast path.

Happy to open a PR guarding the fast path the same way the slow path already is, leaving `split_at` alone.

class SplitEmptyMaxsplitTests(TestCase):
    """Empty iterables yield nothing, whatever ``maxsplit`` is"""

    def test_split_before_empty_any_maxsplit(self):
        pred = lambda item: item == 0
        actual = {
            maxsplit: list(mi.split_before([], pred, maxsplit=maxsplit))
            for maxsplit in (-1, 0, 1, 2)
        }
        expected = {maxsplit: [] for maxsplit in (-1, 0, 1, 2)}
        self.assertEqual(actual, expected)

    def test_split_after_empty_any_maxsplit(self):
        pred = lambda item: item == 0
        actual = {
            maxsplit: list(mi.split_after([], pred, maxsplit=maxsplit))
            for maxsplit in (-1, 0, 1, 2)
        }
        expected = {maxsplit: [] for maxsplit in (-1, 0, 1, 2)}
        self.assertEqual(actual, expected)

    def test_split_when_empty_any_maxsplit(self):
        pred = lambda a, b: a > b
        actual = {
            maxsplit: list(mi.split_when([], pred, maxsplit=maxsplit))
            for maxsplit in (-1, 0, 1, 2)
        }
        expected = {maxsplit: [] for maxsplit in (-1, 0, 1, 2)}
        self.assertEqual(actual, expected)

    def test_split_before_empty_iterator_maxsplit_zero(self):
        pred = lambda item: item == 0
        actual = list(mi.split_before(iter([]), pred, maxsplit=0))
        self.assertEqual(actual, [])

    def test_split_after_empty_iterator_maxsplit_zero(self):
        pred = lambda item: item == 0
        actual = list(
            mi.split_after((item for item in []), pred, maxsplit=0)
        )
        self.assertEqual(actual, [])

    def test_split_when_empty_iterator_maxsplit_zero(self):
        pred = lambda a, b: a > b
        actual = list(mi.split_when(iter([]), pred, maxsplit=0))
        self.assertEqual(actual, [])

    def test_maxsplit_zero_still_yields_whole_iterable(self):
        pred = lambda item: item == 0
        self.assertEqual(
            list(mi.split_before([1, 2, 0, 3], pred, maxsplit=0)),
            [[1, 2, 0, 3]],
        )
        self.assertEqual(
            list(mi.split_after([1, 2, 0, 3], pred, maxsplit=0)),
            [[1, 2, 0, 3]],
        )
        self.assertEqual(
            list(mi.split_when([1, 2, 0, 3], lambda a, b: a > b, maxsplit=0)),
            [[1, 2, 0, 3]],
        )
        self.assertEqual(
            list(mi.split_before(iter([1, 2, 0, 3]), pred, maxsplit=0)),
            [[1, 2, 0, 3]],
        )

    def test_split_at_empty_any_maxsplit_unchanged(self):
        pred = lambda item: item == 0
        actual = {
            maxsplit: list(mi.split_at([], pred, maxsplit=maxsplit))
            for maxsplit in (-1, 0, 1, 2)
        }
        expected = {maxsplit: [[]] for maxsplit in (-1, 0, 1, 2)}
        self.assertEqual(actual, expected)
