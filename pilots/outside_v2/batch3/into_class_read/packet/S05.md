## The issue

Decorator version of serialize()

**Proposal and rationale:**

The `serialize` tool wraps a single iterator instance and is used inline with instance creation: `serialize(itertools.count(100, 5))`.

If every instance needs to be serialized, it is more convenient to wrap the function that creates the iterator: `atomic_counter = synchonized(itertools.count)`.  Later, serialized instances can be created directly: `atomic_counter(100, 5)`.

The wrapper can also be used as a decorator around a generator definition.

**Draft implmentation:**

```python
def synchronized(func):
    """Wrap an function that produces iterators to serialized all instances.

    Existing itertools and more-itertools can be wrapped so that their
    instances are serialized.

    For example, ``itertools.count`` does not make thread-safe instances
    but that is easily fixed with::

        atomic_counter = synchronized(itertools.count)

    Can also be used as a decorator for generator functions definitions
    so that the generator instances are serialized::

        @synchronized
        def enumerate_and_timestamp(iterable):
            for count, value in enumerate(iterable):
                yield count, time_ns(), value                

    """
    @functools.wraps(func)
    def inner(*args, **kwargs):
        iterator = func(*args, **kwargs)
        return serialize(iterator)
    return inner
```

**Notes:**

The name "synchronized" came from Java.

class TestSynchronized(TestCase):
    def test_concurrent_calls(self):
        unique = 10  # Number of distinct counters
        repetitions = 5  # Number of times each counter is used
        limit = 100  # Calls per counter per repetition

        @mi.synchronized
        def atomic_counter():
            # This is a generator so that non-concurrent calls are detectable.
            # To make calls while running more likely, this code uses random
            # time delays.
            i = 0
            while True:
                yield i
                next_i = i + 1
                sleep(random() / 1000)
                i = next_i

        def consumer(counter):
            for i in range(limit):
                next(counter)

        unique_counters = [atomic_counter() for _ in range(unique)]
        counters = unique_counters * repetitions
        workers = [
            Thread(target=consumer, args=[counter]) for counter in counters
        ]
        for worker in workers:
            worker.start()
        for worker in workers:
            worker.join()
        self.assertEqual(
            {next(counter) for counter in unique_counters},
            {limit * repetitions},
        )

class TestSynchronized(TestCase):
    """Tests for ``synchronized()``"""

    def test_wraps_iterator_producing_function(self):
        try:
            synchronized = mi.synchronized
        except AttributeError as exc:
            raise AssertionError(
                'more_itertools.synchronized is not available'
            ) from exc

        atomic_counter = synchronized(count)
        first = atomic_counter(100, 5)
        second = atomic_counter(1000, 7)

        self.assertIsInstance(first, mi.serialize)
        self.assertIsInstance(second, mi.serialize)
        self.assertIsNot(first, second)
        self.assertIsNot(first.lock, second.lock)

        self.assertEqual([next(first) for _ in range(3)], [100, 105, 110])
        self.assertEqual([next(second) for _ in range(3)], [1000, 1007, 1014])
        self.assertEqual(next(first), 115)

    def test_as_decorator_of_generator_function(self):
        try:
            synchronized = mi.synchronized
        except AttributeError as exc:
            raise AssertionError(
                'more_itertools.synchronized is not available'
            ) from exc

        @synchronized
        def enumerate_values(iterable):
            for position, value in enumerate(iterable):
                yield position, value

        first = enumerate_values('ABC')
        second = enumerate_values('xyz')

        self.assertIsInstance(first, mi.serialize)
        self.assertIsInstance(second, mi.serialize)
        self.assertEqual(next(second), (0, 'x'))
        self.assertEqual(list(first), [(0, 'A'), (1, 'B'), (2, 'C')])
        self.assertEqual(list(second), [(1, 'y'), (2, 'z')])

    def test_exhausted_instance_stays_stopped(self):
        try:
            synchronized = mi.synchronized
        except AttributeError as exc:
            raise AssertionError(
                'more_itertools.synchronized is not available'
            ) from exc

        wrapped = synchronized(iter)([1, 2])

        self.assertIs(iter(wrapped), wrapped)
        self.assertEqual(list(wrapped), [1, 2])
        self.assertEqual(list(wrapped), [])

    def test_shared_instance_across_threads(self):
        try:
            synchronized = mi.synchronized
        except AttributeError as exc:
            raise AssertionError(
                'more_itertools.synchronized is not available'
            ) from exc

        result = 0
        result_lock = Lock()

        @synchronized
        def producer(limit):
            'Non-concurrent producer. A generator version of range(limit).'
            for value in range(limit):
                yield value

        def consumer(counter):
            nonlocal result
            total = 0
            for value in counter:
                total += value
            with result_lock:
                result += total

        limit = 10**5
        counter = producer(limit)
        workers = [Thread(target=consumer, args=[counter]) for _ in range(8)]
        for worker in workers:
            worker.start()
        for worker in workers:
            worker.join()

        self.assertEqual(result, limit * (limit - 1) // 2)
